###### 牺牲一个存储单元-队列-代码 🤯 

```c++
#include <stdio.h>
#include <stdlib.h>

#define MaxSize 5
#define ElemType int

typedef struct {
    ElemType data[MaxSize];
    int front, rear;
} SqQueue;

void InitSeQueue(SqQueue &Q) {
    Q.front = Q.rear = 0;
}

bool EnQueue(SqQueue &Q, ElemType x) {
    if ((Q.rear + 1) % MaxSize == Q.front) {
        return false;
    };
    Q.data[Q.rear] = x;
    Q.rear = (Q.rear + 1) % MaxSize;
    return true;
}

bool DeQueue(SqQueue &Q, ElemType &del) {
    if (Q.rear == Q.front) {
        return false;
    }
    del = Q.data[Q.front];
    Q.front = (Q.front + 1) % MaxSize;
    return true;
}

int main() {
    SqQueue Q;
    InitSeQueue(Q);
    int del;

    EnQueue(Q, 111);
    EnQueue(Q, 222);
    EnQueue(Q, 333);
    EnQueue(Q, 444);
    EnQueue(Q, 555);
    EnQueue(Q, 666);

    DeQueue(Q, del);
    DeQueue(Q, del);
    DeQueue(Q, del);
    DeQueue(Q, del);
    DeQueue(Q, del);


}
```

```c++
#include <stdio.h>
#include <stdlib.h>

#define MaxSize 5
#define ElemType int

typedef struct {
    ElemType data[MaxSize];
    int front, rear;
    int tag;
} SqQueue;

void InitSeQueue(SqQueue &Q) {
    Q.front = Q.rear = 0;
    Q.tag = 0;
}

bool EnQueue(SqQueue &Q, ElemType x) {
    if (Q.rear == Q.front && Q.tag == 1) {
        return false;
    };
    Q.data[Q.rear] = x;
    Q.rear = (Q.rear + 1) % MaxSize;
    Q.tag = 1;
    return true;
}

bool DeQueue(SqQueue &Q, ElemType &del) {
    if (Q.rear == Q.front && Q.tag == 0) {
        return false;
    }
    del = Q.data[Q.front];
    Q.front = (Q.front + 1) % MaxSize;
    Q.tag = 0;
    return true;
}


int main() {
    SqQueue Q;
    InitSeQueue(Q);
    int del;
    for (int i = 1; i <= 5; i++) {
        EnQueue(Q, i * 10);
//        printf("sum=%d\n", i * 10);
    }
    DeQueue(Q, del);


}
```

好，那和之前讲的类似，如果我们让所有的地方都填满数据元素的话。那这种情况下，

<u>那处理方法和之前一样，</u>
我们可以规定头指针前面的这个存储单元不可以存放数据元素，
那就意味着尾指针最多只有可能指向这个位置。
这样的话，在队满和队空的时候，

*而如果头指针是在尾指针后面的两个位置，那就说明此时队列是满的。*
