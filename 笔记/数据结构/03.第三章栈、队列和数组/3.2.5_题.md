######  🤯 判断队列为空的条件-题型

就是我们假设只有一个元素的时候，会怎么样，就是现在这样

![image-20250216170626564](/Users/yuebinghui/Documents/program/github/note/images/image-20250216170626564.png)

然后去倒着，推这个队列为空的时候，
那么我们想<u>队列为空的时候</u>*到*<u>队列当中只有一个元素</u>，
那么它所要完成的操作就是有元素要进队嘛，

那么有元素要进队，我们知道队头指针肯定是不动的，
那么应该是队尾指针。向下标增大的方向移动

所以现在队尾指针是指向下标为零的这个地方的，
<u>那么在队尾指针没有移动之前，它应该是指向哪里呢？</u>
那么应该是指向这里。就是maxsize- 1也就是它和front指向的呢，是同一个地方，

那么只有是这样的一种初始情况，它队列为空

![image-20250216170938641](/Users/yuebinghui/Documents/program/github/note/images/image-20250216170938641.png)

就是你有一个元素的时候，你的指针是 front在maxsize-1，rear在0
我回退一步，当我为空的时候，就是rear在maxsize-1，
那我入队就是rear+1，然后rear就变成0了，就是题目中的情况了，也就是队列里有一个元素

```c++
#include <stdio.h>
#include <stdlib.h>

#define MaxSize 5
#define ElemType int

typedef struct {
    ElemType data[MaxSize];
    int front, rear;
    int size;
} SqQueue;

typedef struct {
    ElemType data[MaxSize];
    int top;
} SqStack;

// 栈代码👇🏻
void InitStack(SqStack &S) {
    S.top = -1;
}

bool Push(SqStack &S, ElemType x) {
    if (S.top == MaxSize - 1) {
        return false;
    }
    S.data[++S.top] = x;
    return true;
}

bool Pop(SqStack &S, ElemType &del) {
    if (S.top == -1) {
        return false;
    }
    del = S.data[S.top--];
    return true;
}

bool StackEmpty(SqStack S) {
    return S.top == -1;
}
// 栈代码👆🏻

// 队列代码👇🏻
void InitSeQueue(SqQueue &Q) {
    Q.front = Q.rear = 0;
    Q.size = 0;
}

bool EnQueue(SqQueue &Q, ElemType x) {
    if (Q.size == MaxSize) {
        return false;
    };
    Q.data[Q.rear] = x;
    Q.rear = (Q.rear + 1) % MaxSize;
    Q.size += 1;
    return true;
}

bool DeQueue(SqQueue &Q, ElemType &del) {
    if (Q.size == 0) {
        return false;
    }
    del = Q.data[Q.front];
    Q.front = (Q.front + 1) % MaxSize;
    Q.size -= 1;
    return true;
}

bool GetHead(SqQueue Q, ElemType &x) {
    if (Q.size == 0) {
        return false;
    }
    x = Q.data[Q.front];
    return true;
}

bool QueueStack(SqQueue Q) {
    return Q.size == 0;
}
// 队列代码👆🏻


int main() {
    SqQueue Q;
    InitSeQueue(Q);

    SqStack S;
    InitStack(S);

    int del;
    for (int i = 1; i <= 5; i++) {
        EnQueue(Q, i * 10);
//        printf("sum=%d\n", i * 10);
    }

    while (!QueueStack(Q)) {
        DeQueue(Q, del);
        Push(S, del);
    }

    while (!StackEmpty(S)) {
        Pop(S, del);
        EnQueue(Q, del);
    }

    while (!QueueStack(Q)) {
        DeQueue(Q, del);
        printf("sum=%d\n", del);
    }


}
```
