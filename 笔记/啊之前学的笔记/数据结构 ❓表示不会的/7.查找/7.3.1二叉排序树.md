    if (NULL == T) {
        T = (BSTree) malloc(sizeof(BSTNode));
        T->key = k;
        T->lchild = T->rchild = NULL;
        return 1;
    }
    BSTNode *p = T, *parent;
    while (p) {
        parent = p;
        if (p->key == k) {
            return 0;
        } else if (p->key > k) {
            p = p->lchild;
        } else {
            p = p->rchild;
        }
    }

    BSTNode *NewT = (BSTree) malloc(sizeof(BSTNode));
    NewT->key = k;
    NewT->lchild = NewT->rchild = NULL;

    if (parent->key > k) {
        parent->lchild = NewT;
    } else {
        parent->rchild = NewT;
    }
    return 1;
}

void CreateBSTree(BSTree &T, int arr[], int n) {
    T = NULL;
    int i = 0;
    while (i < n) {
        BSTInsert(T, arr[i]);
        i++;
    }
}

}

void DeleteBSTree(BSTree &T, ElemType key) {
    if (T == NULL) {
        return;
    }
    if (key > T->key) {
        DeleteBSTree(T->rchild, key);
    } else if (key < T->key) {
        DeleteBSTree(T->lchild, key);
    } else {
        if (T->lchild == NULL) {
            BSTNode *tempNode = T;
            T = T->rchild;
            free(tempNode);
        } else if (T->rchild == NULL) {
            BSTNode *tempNode = T;
            T = T->lchild;
            free(tempNode);
        } else {
            BSTNode *tempNode = T->rchild; // 右子树的结点 找到右子树的最左结点
            while (tempNode->lchild) {
                tempNode = tempNode->lchild;
            }
            T->key = tempNode->key;
            DeleteBSTree(T->rchild, tempNode->key);
        }
    }

}
